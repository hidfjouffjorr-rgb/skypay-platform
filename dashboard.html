const express = require('express'); const { Pool } = require('pg'); const path = require('path'); const https = require('https'); const app = express(); app.use(express.json()); app.use(express.static(path.join(__dirname, '/'))); // Database Connection const pool = new Pool({ connectionString: process.env.DATABASE_URL, ssl: process.env.DATABASE_URL ? { rejectUnauthorized: false } : false }); // Telegram Config const TELEGRAM_BOT_TOKEN = process.env.TELEGRAM_BOT_TOKEN || 'YOUR_TELEGRAM_BOT_TOKEN_HERE'; const TELEGRAM_CHAT_ID = '6804193991'; function sendTelegramMessage(text) { if (TELEGRAM_BOT_TOKEN === 'YOUR_TELEGRAM_BOT_TOKEN_HERE') return; const data = JSON.stringify({ chat_id: TELEGRAM_CHAT_ID, text: text, parse_mode: 'HTML' }); const options = { hostname: 'api.telegram.org', port: 443, path: `/bot${TELEGRAM_BOT_TOKEN}/sendMessage`, method: 'POST', headers: { 'Content-Type': 'application/json', 'Content-Length': data.length } }; const req = https.request(options, res => { res.on('data', () => {}); }); req.on('error', err => console.error('Telegram API Error:', err)); req.write(data); req.end(); } // Global Settings Store let appSettings = { bkash: '', nagad: '', min_deposit: '300', min_withdraw: '500', package_name: 'VIP Starter', package_duration: '30', apk_url: '' }; // API Endpoints app.get('/api/settings', (req, res) => { res.json({ success: true, settings: appSettings }); }); app.post('/api/settings', (req, res) => { appSettings = { ...appSettings, ...req.body }; res.json({ success: true, message: 'Settings updated successfully' }); }); // Deposit Request API app.post('/api/deposit', (req, res) => { const { phone, amount, trxId } = req.body; const msg = `📥 নতুন ডিপোজিট রিকোয়েস্ট!\n\n👤 ইউজার: ${phone}\n💵 পরিমাণ: ৳${amount}\n🧾 TrxID: ${trxId}`; sendTelegramMessage(msg); res.json({ success: true, message: 'ডিপোজিট রিকোয়েস্ট টেলিগ্রামে পাঠানো হয়েছে!' }); }); // Withdraw Request API app.post('/api/withdraw', (req, res) => { const { phone, amount, targetNumber } = req.body; const msg = `📤 নতুন উইথড্র রিকোয়েস্ট!\n\n👤 ইউজার: ${phone}\n💵 পরিমাণ: ৳${amount}\n📱 প্রাপক নম্বর: ${targetNumber}`; sendTelegramMessage(msg); res.json({ success: true, message: 'উইথড্র রিকোয়েস্ট টেলিগ্রামে পাঠানো হয়েছে!' }); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
